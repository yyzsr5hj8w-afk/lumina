class BattleSystem{
constructor(game){this.game=game;this.enemy=null;this.bossId=null;this.active=false;this.busy=false;this.guard=false}
start(enemy,bossId=null){this.enemy=JSON.parse(JSON.stringify(enemy));this.bossId=bossId;this.active=true;this.busy=false;this.guard=false;this.game.mode="battle";battleUI.classList.add("show");battleSub.classList.remove("show");bossId?audio.boss():audio.battle();this.log(`${this.enemy.name} が あらわれた！`)}
log(t){battleLog.textContent=t}
wait(ms){return new Promise(r=>setTimeout(r,ms))}
dmg(atk,def,m=1){return Math.max(1,Math.floor(atk*m-def*.5+rnd(-2,2)))}
async action(type,payload){
if(!this.active||this.busy)return;
this.busy=true;this.guard=false;
if(type==="attack")await this.attack();
else if(type==="magic"){this.showMagic();this.busy=false;return}
else if(type==="skill"){this.showSkills();this.busy=false;return}
else if(type==="item"){this.showItems();this.busy=false;return}
else if(type==="guard"){this.guard=true;this.log(`${this.game.player.name}は 防御の構えをとった！`);await this.wait(600);await this.enemyTurn()}
else if(type==="run")await this.run();
else if(type==="magicUse")await this.useMagic(payload);
else if(type==="skillUse")await this.useSkill(payload);
else if(type==="itemUse")await this.useItem(payload);
this.busy=false;
}
async attack(){
const p=this.game.player,crit=Math.random()<.1,d=this.dmg(p.atk,this.enemy.def,crit?1.9:1);
this.enemy.hp-=d;audio.attack();this.game.enemyShake=10;
this.log(crit?`会心の一撃！\n${this.enemy.name}に ${d} のダメージ！`:`${p.name}のこうげき！\n${this.enemy.name}に ${d} のダメージ！`);
await this.wait(650);if(await this.deadCheck())return;await this.enemyTurn();
}
showMagic(){battleSub.innerHTML="";const cost={フレイム:3,ヒール:3,スパーク:5,エリアヒール:7,セイントレイ:9};this.game.player.magics.forEach(m=>{const b=document.createElement("button");b.className="uiBtn";b.textContent=`${m} MP${cost[m]}`;b.onclick=()=>this.action("magicUse",m);battleSub.appendChild(b)});battleSub.classList.add("show")}
showSkills(){battleSub.innerHTML="";const cost={パワースラッシュ:2,ガードブレイク:3,勇気の盾:3,英雄の一撃:7};this.game.player.skills.forEach(s=>{const b=document.createElement("button");b.className="uiBtn";b.textContent=`${s} MP${cost[s]}`;b.onclick=()=>this.action("skillUse",s);battleSub.appendChild(b)});battleSub.classList.add("show")}
showItems(){battleSub.innerHTML="";Object.entries(this.game.player.items).forEach(([name,n])=>{if(n<=0||!ITEMS[name]||ITEMS[name].type==="quest")return;const b=document.createElement("button");b.className="uiBtn";b.textContent=`${name} ×${n}`;b.onclick=()=>this.action("itemUse",name);battleSub.appendChild(b)});battleSub.classList.add("show")}
async useMagic(name){
battleSub.classList.remove("show");const p=this.game.player,cost={フレイム:3,ヒール:3,スパーク:5,エリアヒール:7,セイントレイ:9}[name];
if(p.mp<cost){this.log("MPが足りない！");await this.wait(600);return}
p.mp-=cost;audio.magic();
if(name==="ヒール"||name==="エリアヒール"){const heal=name==="ヒール"?18+p.mag:35+p.mag*2;p.hp=Math.min(p.maxHp,p.hp+heal);audio.heal();this.log(`${name}！\nHPが ${heal} 回復した！`)}
else{const mult=name==="フレイム"?1.35:name==="スパーク"?1.7:2.2,d=this.dmg(p.mag+6,this.enemy.def*.5,mult);this.enemy.hp-=d;this.game.enemyShake=12;this.log(`${name}！\n${this.enemy.name}に ${d} の魔法ダメージ！`)}
this.game.updateHUD();await this.wait(700);if(await this.deadCheck())return;await this.enemyTurn();
}
async useSkill(name){
battleSub.classList.remove("show");const p=this.game.player,cost={パワースラッシュ:2,ガードブレイク:3,勇気の盾:3,英雄の一撃:7}[name];
if(p.mp<cost){this.log("MPが足りない！");await this.wait(600);return}
p.mp-=cost;
if(name==="勇気の盾"){p.buff.def=5;this.log("勇気の盾！\n防御力が上がった！")}
else if(name==="ガードブレイク"){this.enemy.def=Math.max(0,this.enemy.def-4);this.log("ガードブレイク！\n敵の防御力を下げた！")}
else{const m=name==="英雄の一撃"?2.4:1.55,d=this.dmg(p.atk,this.enemy.def,m);this.enemy.hp-=d;audio.attack();this.game.enemyShake=13;this.log(`${name}！\n${this.enemy.name}に ${d} のダメージ！`)}
this.game.updateHUD();await this.wait(700);if(await this.deadCheck())return;await this.enemyTurn();
}
async useItem(name){
battleSub.classList.remove("show");const p=this.game.player;if((p.items[name]||0)<=0)return;const item=ITEMS[name];p.items[name]--;
if(item.type==="heal")p.hp=Math.min(p.maxHp,p.hp+item.value);if(item.type==="mp")p.mp=Math.min(p.maxMp,p.mp+item.value);if(item.type==="both"){p.hp=Math.min(p.maxHp,p.hp+item.hp);p.mp=Math.min(p.maxMp,p.mp+item.mp)}
audio.heal();this.game.updateHUD();this.log(`${name}を使った！`);await this.wait(600);await this.enemyTurn();
}
async run(){if(this.bossId){this.log("ボスからは逃げられない！");await this.wait(600);await this.enemyTurn();return}if(Math.random()<.65){this.log("うまく逃げ切った！");await this.wait(500);this.end()}else{this.log("逃げられなかった！");await this.wait(500);await this.enemyTurn()}}
async enemyTurn(){
if(!this.active)return;const p=this.game.player;let d=0;
if(this.bossId==="demon"||this.bossId==="trueDemon"){const phase=this.enemy.phase,r=Math.random();if(phase===2&&r<.22){d=this.dmg(this.enemy.mag,p.def*.4,1.65);this.log(`真・ヴァルグレインの「深淵波」！\n${d} のダメージ！`)}else if(r<.42){d=this.dmg(this.enemy.mag,p.def*.5,1.35);this.log(`${this.enemy.name}の闇魔法！\n${d} のダメージ！`)}else if(phase===2&&r<.58){this.enemy.atk+=2;this.enemy.mag+=2;this.log("真・ヴァルグレインは闇の力を吸収した！")}else{d=this.dmg(this.enemy.atk,p.def,1);this.log(`${this.enemy.name}のこうげき！\n${d} のダメージ！`)}}
else if(this.bossId==="sky"&&Math.random()<.45){d=this.dmg(this.enemy.mag,p.def*.5,1.25);this.log("蒼天の魔導騎士の雷撃！\n"+d+" のダメージ！")}
else if(this.bossId==="forestGuardian"&&Math.random()<.35){d=this.dmg(this.enemy.mag,p.def*.5,1.2);this.log("霧森の魔獣の毒霧！\n"+d+" のダメージ！")}
else{d=this.dmg(this.enemy.atk,p.def,1);this.log(`${this.enemy.name}のこうげき！\n${d} のダメージ！`)}
if(this.guard)d=Math.max(1,Math.floor(d*.45));if(d>0){p.hp-=d;audio.hurt();this.game.screenShake=8;this.game.updateHUD()}
await this.wait(700);if(p.hp<=0){p.hp=0;this.game.updateHUD();this.active=false;await this.game.gameOver()}
}
async deadCheck(){
if(this.enemy.hp>0)return false;
if(this.bossId==="demon"){this.log("魔王ヴァルグレインを倒した……！\nしかし、闇の力が暴走する！");await this.wait(1300);this.enemy=JSON.parse(JSON.stringify(BOSSES.trueDemon));this.bossId="trueDemon";audio.boss();this.log("真・ヴァルグレインが出現した！");return true}
if(this.bossId==="trueDemon"){this.log("真・ヴァルグレインを倒した！");audio.victory();await this.wait(1000);this.end();this.game.story=99;this.game.showEnding();return true}
const p=this.game.player;p.gold+=this.enemy.gold||0;const levels=p.gainExp(this.enemy.exp||0);
if(this.enemy.name==="マッシュビースト")this.game.questKills.mush=(this.game.questKills.mush||0)+1;
this.log(`${this.enemy.name}を倒した！\nEXP ${this.enemy.exp||0} / ${this.enemy.gold||0}G を獲得！`);audio.victory();this.game.updateHUD();await this.wait(800);
if(levels>0){audio.level();this.game.showBanner("LEVEL UP!");this.log(`${p.name}は LV${p.lv} になった！\n能力が上がった！`);await this.wait(1000)}
if(this.bossId){this.game.bossDefeated[this.bossId]=true;if(BOSSES[this.bossId]?.rewardCrystal){this.game.addCrystal();this.game.showBanner("光の結晶を手に入れた！")}}
this.end();this.game.updateObjective();return true;
}
end(){this.active=false;battleUI.classList.remove("show");battleSub.classList.remove("show");this.game.mode="field";this.game.encounterCooldown=5}
}
